⚡ 1. C++ (Arduino Microcontrollers)

// Arduino C++ Digital Output Example
const int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  digitalWrite(ledPin, HIGH);
  delay(1000);
  digitalWrite(ledPin, LOW);
  delay(1000);
}

from machine import Pin
import time

led = Pin(2, Pin.OUT)
distance = 15

if distance < 20:
    print("Obstacle Detected!")
    led.value(1)
else:
    led.value(0)
